% ===== SLIDE 2: Motivation =====
\begin{frame}{Housing Crisis and Local Opposition}
    \begin{columns}
        \column{0.55\textwidth}
        \textbf{The Affordability Problem}
        \begin{itemize}
            \item Austin grew $\approx$40\% between 2010--2024.
            \item New housing requires discretionary rezoning approval.
            \item Projects face public hearings and organized opposition.
        \end{itemize}
        \vspace{0.8em}
        \textbf{The Political Mechanism}

        Texas law gave adjacent property owners a formal institutional veto: the \textbf{protest petition}.

        \column{0.45\textwidth}
        \centering
        \includegraphics[width=\linewidth, height=0.62\textheight, keepaspectratio]{exhibits/mercatus_map.png}
    \end{columns}
\end{frame}
\note{\textbf{[2 min]} Set the stakes. Austin's rapid growth created real housing scarcity, and the political process for approving new housing was structurally vulnerable to organized local opposition. Preview: HB 24 (2025) reformed this regime; we are mapping a specific historical era.}

% ===== SLIDE 3: The Mechanism =====
\begin{frame}{The Protest Petition: An Institutional Veto}
    \begin{columns}
        \column{0.55\textwidth}
        \textbf{Texas LGC Chapter 211: The 20\% Rule}
        \begin{itemize}
            \item Owners of 20\% of land within 200ft may file written protest.
            \item A valid petition triggers a supermajority ($\tfrac{3}{4}$) council vote.
            \item Post-2014 council (11 members): 6 votes $\rightarrow$ 9 required.
            \item 21 states maintain similar petition statutes.
        \end{itemize}
        \vspace{0.8em}
        \textit{A localized minority can block citywide housing decisions.}

        \column{0.45\textwidth}
        \centering
        \includegraphics[width=0.9\textwidth, height=0.60\textheight, keepaspectratio]{appendix/fig_app_04_sample_protest_petition_v15.pdf} \\
        \footnotesize \textit{Sample formal protest petition record}
    \end{columns}
\end{frame}
\note{\textbf{[2 min]} Key counterintuitive point: this is not a citywide referendum, only neighbors within 200 feet. Austin's council expanded from 7 to 11 members with the 10-1 district reform in 2014; the supermajority for 11 members is 9 votes. The petition document shows the precise geometric calculation of signed lot area against total buffer area.}

% ===== SLIDE 4: RD Result =====
\begin{frame}{Why Petitions Matter: Institutional Costs}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Reduced-Form Discontinuity at the 20\% Threshold}
        \begin{itemize}
            \item Running variable: reconstructed petition-share.
            \item Outcome: council processing time.
            \item Estimator: triangular-kernel WLS at the 20\% margin.
        \end{itemize}
        \vspace{0.8em}
        \textbf{Result:} Crossing the threshold adds \textbf{\metricRDDelay{} days} (\metricRDDelayWeeks{} weeks).

        \small 95\% CI: \metricRDCI{} days. McCrary $p > 0.05$.

        \column{0.5\textwidth}
        \centering
        \includegraphics[width=\linewidth, height=0.65\textheight, keepaspectratio]{ch5/fig_ch5_31_petition_rd.png}
    \end{columns}
\end{frame}
\note{\textbf{[3 min]} This answers "so what if we predict petitions?" Crossing 20\% measurably extends processing time by 42.5 days. Be transparent: this is a reduced-form discontinuity, not a sharp RD, because we cannot observe the Clerk's formal certification directly. The key takeaway: this outcome has real institutional consequences and is worth predicting.}

% ===== SLIDE 5: Research Question =====
\begin{frame}{The Research Question}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{The Prediction Problem}
        \vspace{0.5em}

        {\large Can we anticipate protest petitions \textit{before} public hearings using only filing-date records?}

        \vspace{1.2em}
        \textbf{Filing-Date Features}
        \begin{itemize}
            \item Site geometry: lot area, height/FAR deltas.
            \item Current vs.\ requested zoning class.
            \item Buffer demographics and TCAD appraisals.
            \item Historical petition rates in the district.
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{A Sample Case}
        \vspace{1em}
        {\footnotesize
        \begin{tabular}{ll}
            \toprule
            Feature & Case C14-2018-0156 \\
            \midrule
            Gross Site Area & 4.92 Acres \\
            Height Request & +15.5 ft \\
            FAR Request & +0.43 \\
            Neighborhood & Riverside \\
            \textbf{Result} & \textbf{Valid Petition} \\
            \bottomrule
        \end{tabular}
        }
        \vspace{1em}

        \small \textit{Race/ethnicity excluded from predictive features; reserved for post-hoc equity audits only.}
    \end{columns}
\end{frame}
\note{\textbf{[1.5 min]} Stress the as-of constraint: only what a planner knew on the day of filing. The feature partitioning (predictive vs. equity audit) is an important design choice. Sensitive attributes are excluded to reduce proxy discrimination risk, though that risk cannot be fully eliminated.}

% ===== SLIDE 6: Rarity Problem =====
\begin{frame}{The Rarity Problem}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Sample Constraints}
        \begin{itemize}
            \item $N = \metricBaselineCases{}$ cases, 2007--2024.
            \item Base rate: $\approx\metricBaseRate{}$ (1 in 38 cases).
            \item Test-set positives: $n_{\text{pos}} = \metricNTestPositive{}$.
        \end{itemize}
        \vspace{0.8em}
        Sparse positives widen confidence intervals. The output is a \textbf{relative risk ranking}, not a case-level probability.

        \column{0.5\textwidth}
        \textbf{Evaluation Design}
        \begin{itemize}
            \item Train on years $\leq t$; evaluate on years $> t$.
            \item No random splits: mimics forward administrative use.
        \end{itemize}
        \vspace{0.8em}
        \textbf{Primary metric: PR-AUC}

        \small AUROC omitted: 97.4\% true negatives artificially inflate it.
    \end{columns}
\end{frame}
\note{\textbf{[2 min]} PR-AUC over AUROC is the correct metric for this imbalance, not just convention. The temporal holdout is critical: a random split leaks future mobilization patterns into training. Fifteen test-set positives explains the wide CI ([0.48, 0.91]).}

% ===== SLIDE 7: Data Pipeline =====
\begin{frame}{Timeline Integrity: As-Of Features}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Enforcing the Constraint}
        \begin{itemize}
            \item 15+ years of zoning agendas scraped from PDF clerk records.
            \item Filing dates back-calculated from statutory minimums.
            \item ACS and TCAD records anchored to publication date.
            \item $N=8$ IRB-approved interviews (ACYY0820) for institutional grounding.
        \end{itemize}

        \column{0.5\textwidth}
        \centering
        \includegraphics[width=0.9\textwidth, height=0.65\textheight, keepaspectratio]{appendix/fig_app_02_project_overview.pdf}
    \end{columns}
\end{frame}
\note{\textbf{[2 min]} A planner in 2012 could not access 2020 Census data; the feature matrix enforces this. Municipal IT overwrites historical timestamps, so milestone dates are approximated from statutory minimums, introducing measurement error reported as a limitation. Interviews provide context, not model validation.}

% ===== SLIDE 9: Top-Decile Lift =====
\begin{frame}{Operational Utility: Triage in Practice}
    \begin{columns}
        \column{0.55\textwidth}
        \textbf{Top-Decile Risk Ranking}
        \begin{itemize}
            \item Flag the highest-risk 10\% of cases.
            \item Precision: \textbf{\metricTopDecilePrecision{}} (CI: 16.1\%--37.8\%).
            \item Lift: \textbf{\metricTopDecileLift{}} over the base rate.
            \item Captures \metricTopDecileTP{} true positives in \metricTopDecileN{} flagged cases.
        \end{itemize}
        \vspace{0.8em}
        Reviewing flagged cases is \textbf{nearly 10$\times$} more likely to surface a genuine petition case.

        \column{0.45\textwidth}
        \textbf{Appropriate Uses}
        \begin{itemize}
            \item[\checkmark] Prioritize outreach for high-risk cases.
            \item[\checkmark] Flag cases for early negotiation.
            \item[$\times$] Hard probability cutoffs.
            \item[$\times$] Developer site-screening.
        \end{itemize}
        \vspace{0.8em}
        \footnotesize \textit{Ethics Warning: This model ranks risk. It should not be used to pre-screen sites for developers.}
    \end{columns}
\end{frame}
\note{\textbf{[2 min]} The lift metric converts abstract PR-AUC into something actionable. The wide precision CI (16--38\%) is why the use case is triage ranking, not hard thresholds. Raise the ethics warning proactively.}

% ===== SLIDE 11: What Drives Petition Risk =====
\begin{frame}{What Drives Petition Risk}
    \begin{columns}
        \column{0.4\textwidth}
        \textbf{Key Finding}
        \vspace{0.5em}

        Tree models allocate over 80\% of weight to \textbf{socio-demographic} clusters: neighborhood demographics, income, and housing tenure.

        \vspace{0.8em}
        Physical parcel attributes matter far less than \textbf{who lives nearby}.

        \vspace{0.8em}
        Deep models invert this, concentrating weight on parcel scale and valuation.

        \column{0.6\textwidth}
        \centering
        \includegraphics[height=0.68\textheight, width=\linewidth, keepaspectratio]{exhibits/fig_feature_importance_clustered_H0_Full.pdf}
    \end{columns}
\end{frame}
\note{\textbf{[2 min]} Lead with the result: it is who lives nearby that matters most, not what the parcel looks like. Tree ensembles concentrate weight on socio-demographic clusters at over 80\% of total importance. Deep architectures shift toward physical attributes (parcel scale, valuation, structure age). This architecture-dependent split is the key finding on the next slide.}

% ===== SLIDE 12: Architecture Determines Feature Reliance =====
\begin{frame}{Architecture Determines Feature Reliance}
    \begin{columns}
        \column{0.38\textwidth}
        \textbf{Architecture-Dependent Reliance}
        \begin{itemize}
            \item Trees: $>$80\% weight on socio-demographic clusters.
            \item Deep models: shift toward physical and geographic attributes.
            \item Same data, same task: different internal logic.
        \end{itemize}
        \vspace{0.8em}
        \textit{Algorithm selection is a normative choice about which features are prioritized.}

        \column{0.62\textwidth}
        \centering
        \includegraphics[height=0.70\textheight, width=\linewidth, keepaspectratio]{ch5/fig_ch5_35_meta_attribution_clustermap.pdf}
    \end{columns}
\end{frame}
\note{\textbf{[2.5 min]} Rows are model-year combinations, columns are semantic clusters. The Ward-linkage dendrogram separates trees from deep models on the vertical axis; that separation is the finding. Deploying CatBoost endorses socio-demographic context as the primary signal; deploying a deep network endorses physical geometry. This is a policy choice disguised as a technical choice.}

% ===== SLIDE 13: Temporal Drift and Agency (merged) =====
\begin{frame}{Temporal Drift and Neighborhood Agency}
    \begin{columns}
        \column{0.45\textwidth}
        \textbf{When Does the Model Break?}
        \begin{itemize}
            \item Linear models collapse at T+3 horizons and beyond.
            \item Tree ensembles degrade more gracefully.
        \end{itemize}
        \vspace{0.5em}
        \textbf{The 2022 Dip}
        \begin{itemize}
            \item PR-AUC drops sharply across all families in 2022.
            \item Consistent with pandemic-era filing disruptions reaching the evaluation window.
            \item Not a causal claim.
        \end{itemize}
        \vspace{0.5em}
        \small Performance sensitivity to ambient shocks is \textit{consistent with} mobilization being a strategic social choice.

        \column{0.55\textwidth}

% ===== SLIDE 14: Signal Dilution =====
\begin{frame}{Signal Dilution: When Retraining Hurts}
    \begin{columns}
        \column{0.45\textwidth}
        \textbf{The Paradox}
        \begin{itemize}
            \item Pre-2018 models achieve $\sim$34$\times$ lift in 2024.
            \item Retraining on modern data degrades lift to $\sim$19$\times$.
        \end{itemize}
        \vspace{0.8em}
        Post-ordinance labels starve the model of positive examples. The frozen legacy model preserves spatial sorting intelligence from an active institutional era.

        \column{0.55\textwidth}
        \centering
        \includegraphics[width=\linewidth, height=0.65\textheight, keepaspectratio]{exhibits/fig_temporal_drift_lift_testyear.pdf}
    \end{columns}
\end{frame}
\note{\textbf{[2 min]} Retraining actively hurts: exposing the model to the post-ordinance era stops it from penalizing aggressive spatial configurations because the institutional environment has stopped producing petitions. Implication: continuous retraining would lose predictive power precisely as the policy environment stabilizes.}

% ===== SLIDE 15: Qualitative Context =====
\begin{frame}{Qualitative Context: Motive Blindness}
    \begin{columns}
        \column{0.6\textwidth}
        \textbf{What the Algorithm Cannot Distinguish}
        \begin{itemize}
            \item Exclusionary NIMBYism (affluent West Austin blocking density).
            \item Anti-displacement resistance (East Austin protecting tenants).
        \end{itemize}
        Same risk weight; opposite social meaning.

        \vspace{0.8em}
        \textbf{Practitioner Perspectives (IRB ACYY0820, $N=8$)}
        \begin{itemize}
            \item \textit{``Haters are overrepresented and silent supporters are underrepresented.''} (Beaudet)
            \item Rigid cutoffs fail to identify communities in actual need. (Torrado)
            \item Probability thresholds are \textit{``ethically hazardous.''} (Tomko)
        \end{itemize}

        \column{0.4\textwidth}
        \centering
        \includegraphics[width=\linewidth, height=0.60\textheight, keepaspectratio]{appendix/fig_app_01_interview_guide.pdf} \\
        \footnotesize \textit{IRB-Approved Interview Protocol}
    \end{columns}
\end{frame}
\note{\textbf{[2.5 min]} Qualitative and quantitative evidence converge here. Interviews provide institutional context for the performance ceiling, not model validation. Motive blindness is not solvable with more data or better algorithms. Formal opposition data over-represents loud voices because participation carries social costs (Tomko).}

% ===== SLIDE 16: Limitations =====
\begin{frame}{Limitations}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Methodological}
        \begin{itemize}
            \item \textbf{Chilling effects}: Only filed cases are observed; pre-emptive withdrawals are invisible. IPW failed overlap diagnostics.
            \item \textbf{Timestamp imputation}: Milestone dates approximated from statutory minimums.
            \item \textbf{Unobserved certification}: Outcome is reconstructed area share, not the clerk's legal determination.
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Ethical and Governance}
        \begin{itemize}
            \item \textbf{Proxy discrimination}: Neighborhood income and demographics encode class and race dynamics even without protected-class inputs.
            \item \textbf{Motive blindness}: Cannot separate exclusionary NIMBYism from anti-displacement resistance.
            \item \textbf{External validity}: Parameters are Austin-specific and pre-HB~24 only.
        \end{itemize}
    \end{columns}
\end{frame}
\note{\textbf{[2 min]} Address proactively. The chilling-effect limitation is the most substantive: predicting on a censored subset of the decision space. Proxy discrimination cannot be fully resolved by excluding protected-class features. The study maps these as boundaries, not solved problems.}

% ===== SLIDE 17: Conclusion =====
\begin{frame}{Conclusion}
    \small
    \textbf{What we showed}
    \begin{itemize}
        \item Crossing 20\% adds \metricRDDelay{} days of processing time: the outcome is worth predicting.
        \item Filing-date data ranks petition risk above chance (PR-AUC = \metricBootstrapFiling{}, CI: \metricBootstrapFilingCI{}), with top-decile lift of \metricTopDecileLift{}.
        \item Algorithm choice determines which feature clusters drive predictions, a normative decision.
    \end{itemize}

    \textbf{What surprised us}
    \begin{itemize}
        \item Retraining on modern data degrades lift; legacy models preserve spatial sorting intelligence.
        \item Performance ceilings are consistent with mobilization as a strategic social choice.
    \end{itemize}

    \textbf{What it implies}
    \begin{itemize}
        \item Predictions belong in relative triage rankings, not case-level probability thresholds.
        \item Methodology is portable; Austin-specific parameters are not.
    \end{itemize}
\end{frame}
\note{\textbf{[2 min]} Connect back to the opening arc: institutional costs established first (RD), then prediction capability characterized, then limits mapped. Real signal, real limits. Lead with the RD result; it anchors the "so what."}

% ===== SLIDE 18: Policy Implications =====
\begin{frame}{Implications: A Now-Closed Policy Window}
    \textbf{The HB 24 Shift (September 1, 2025)}
    \begin{itemize}
        \item Texas HB 24 exempted residential upzonings from the supermajority requirement.
        \item The 20\% institutional veto no longer applies to the most contested case types.
        \item This thesis maps a specific, now-closed historical era of minority veto power.
    \end{itemize}

    \vspace{1.2em}
    \textbf{What travels forward}
    \begin{itemize}
        \item Methodology (time-stamped features, temporal holdouts, semantic clustering, qualitative integration) is portable to other discretionary review systems.
        \item Iowa, Oklahoma, and Arizona retain similar petition statutes. The framework applies wherever institutional veto mechanisms create predictable opposition patterns.
    \end{itemize}
\end{frame}
\note{\textbf{[1.5 min]} This is not a live policy problem; it is a historical analysis of how a specific mechanism operated. The analytical contribution is the methodology. Iowa (20\% at 200ft), Oklahoma (tiered), and Arizona (20\% land plus 20\% lots) retain comparable statutes. The framework is designed to travel.}
