% ===== SLIDE 3: The Mechanism (~2 min) =====
\begin{frame}{The Protest Petition: An Institutional Veto}
    \begin{columns}
        \column{0.55\textwidth}
        \textbf{Texas LGC Chapter 211 --- The 20\% Rule}
        \begin{itemize}
            \item Owners of \textbf{20\% of land area} within 200ft of a proposed rezoning may file written protest.
            \item A valid petition forces a \textbf{supermajority vote} ($\tfrac{3}{4}$ of council) instead of a simple majority.
            \item On Austin's 7-member council: 4 votes $\rightarrow$ 6 votes needed for approval.
            \item 21 states have similar architectures (Furth \& McKinley 2025).
        \end{itemize}
        \vspace{0.8em}
        \textit{A localized minority of adjacent owners can effectively block citywide housing decisions.}

        \column{0.45\textwidth}
        \centering
        \includegraphics[width=0.9\textwidth]{appendix/fig_app_04_sample_protest_petition_v15.pdf} \\
        \footnotesize \textit{Sample formal protest petition record}
    \end{columns}
\end{frame}
\note{\textbf{[~2 min]} Walk through the mechanics carefully. The key counterintuitive point: this is not a referendum of the whole city --- only neighbors within 200 feet. Emphasize the asymmetry: a localized area minority enforces a supermajority constraint on city-wide legislative bodies. Show the petition document: it is a precise geometric calculation of signed lot area against total buffer area.}

% ===== SLIDE 6: Rarity Problem + Evaluation Design (~2 min) =====
\begin{frame}{The Rarity Problem}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Sample Constraints}
        \begin{itemize}
            \item \textbf{Universe}: 2007--2024 Austin discretionary rezonings.
            \item \textbf{Analytic sample}: $N = \metricBaselineCases{}$ cases after exclusions.
            \item \textbf{Petition base rate}: $\approx\metricBaseRate{}$ --- fewer than 1 in 38 cases.
            \item \textbf{Test-set positives}: Only $n_{\text{pos}} = \metricNTestPositive{}$ petitioned cases in the temporal holdout.
        \end{itemize}
        \vspace{0.8em}
        Sparse positives widen confidence intervals and prevent reliable case-level probability use.

        The appropriate output is a \textbf{relative risk ranking}, not a probability score for individual decisions.

        \column{0.5\textwidth}
        \textbf{Evaluation: Strictly Temporal Holdouts}
        \begin{itemize}
            \item No random train/test splits.
            \item Train on years $\leq t$; evaluate on years $> t$.
            \item Mimics forward administrative use.
        \end{itemize}
        \vspace{0.8em}
        \textbf{Primary metric: Precision-Recall AUC}

        \small AUROC is omitted: with $\approx$97.4\% true negatives, the large denominator artificially inflates the curve and obscures minority-class performance. PR-AUC forces the model to balance recall against the precision cost of false alarms.
    \end{columns}
\end{frame}
\note{\textbf{[~2 min]} The rarity problem is the central methodological challenge. Emphasize: PR-AUC over AUROC is not just a convention --- it is the correct metric for this class imbalance. The temporal holdout design is critical for realism; a random split would leak information about future mobilization patterns into training. The $N=15$ test positives explains why the CI is wide ([0.48, 0.91]) --- this is honest uncertainty, not a modeling failure.}

        \column{0.55\textwidth}

        \begin{itemize}
            \item Flag the \textbf{highest-risk 10\%} of incoming cases.
            \item Precision in that tier: \textbf{\metricTopDecilePrecision{}} (CI: 16.1\%--37.8\%).

            \item True positives captured: \metricTopDecileTP{} of \metricTopDecileN{} flagged cases.
        \end{itemize}
        \vspace{0.8em}
        \textbf{In plain terms:} A planner reviewing the top-decile flagged cases is nearly \textbf{10$\times$} more likely to encounter a genuine petition case than by random case selection.

        \column{0.45\textwidth}
        \textbf{The Appropriate Use Case}
        \vspace{0.5em}
        \begin{itemize}
            \item[\checkmark] \textbf{Triage}: Prioritize outreach and early negotiation for high-risk cases.
            \item[\checkmark] \textbf{Monitoring}: Flag cases that historically co-occur with petition mobilization.
        \end{itemize}
        \vspace{0.5em}
        \begin{itemize}
            \item[$\times$] \textbf{Not}: Deterministic probability thresholds for case-level go/no-go decisions.
            \item[$\times$] \textbf{Not}: Developer site-screening to avoid high-risk neighborhoods.
        \end{itemize}
        \vspace{0.5em}
        \footnotesize \textit{Ethics Warning: This model ranks risk. It should not be used to pre-screen sites for developers seeking the path of least resistance.}
    \end{columns}
\end{frame}
\note{\textbf{[~2 min]} This is the most important slide for practitioners on the committee. The lift metric converts abstract PR-AUC into something actionable: reviewing the top-decile flagged cases gives roughly 10x the signal density of random selection. The precision CI (16--38\%) remains wide, which is why the appropriate use is triage ranking, not hard thresholds. Raise the ethics warning proactively --- this pre-empts the committee from asking whether the tool could be weaponized against vulnerable neighborhoods.}

% ===== SLIDE 11: Collinearity / Semantic Clustering (~2 min) =====
\begin{frame}{Dealing with Collinearity: Semantic Feature Clustering}
    \begin{columns}
        \column{0.4\textwidth}
        \textbf{The Problem}
        \vspace{0.5em}

        Administrative real estate records contain highly correlated proxies: site area, deed acreage, and structure square footage all measure the same physical concept.

        Standard SHAP attribution dilutes importance arbitrarily across collinear variables.
        \vspace{0.8em}

        \textbf{The Solution}

        Group variables into semantic clusters (Spearman $|r| > 0.70$), then \textbf{sum absolute TreeSHAP values} within each cluster. Pre-clustering before training loses information; post-hoc aggregation preserves model fidelity.

        \column{0.6\textwidth}
        \centering
        \includegraphics[height=0.75\textheight, width=\linewidth, keepaspectratio]{exhibits/fig_feature_importance_clustered_H0_Full.pdf}
    \end{columns}
\end{frame}
\note{\textbf{[~2 min]} The semantic clustering is both a methodological necessity and an interpretive contribution. Key finding in the figure: tree ensembles concentrate reliance on socio-demographic clusters (Demographics, Income, Housing Tenure $>$ 80\% of total weight), while deep architectures re-weight heavily toward physical and geographic attributes (Parcel Scale, Property Valuation, Structure Age). This architecture-dependent reliance will be the key finding on the next slide.}

% ===== SLIDE 14: Temporal Drift --- Lift / Signal Dilution (~2 min) =====
\begin{frame}{Signal Dilution: When Retraining Hurts}
    \begin{columns}
        \column{0.45\textwidth}
        \textbf{The Paradox}
        \vspace{0.5em}
        \begin{itemize}
            \item When petition rates fall post-ordinance, absolute PR-AUC declines even if ranking is perfect. \textbf{Lift} (PR-AUC $\div$ base rate) corrects for this.
            \item Models trained on \textit{pre-2018 data} achieve $\sim$34$\times$ lift in 2024.
            \item Retraining on modern data \textit{degrades} lift to $\sim$19$\times$.
        \end{itemize}
        \vspace{0.8em}
        \textbf{Why:} Post-ordinance labels starve the model of positive examples. The frozen legacy model preserves spatial sorting intelligence learned when the institutional environment was active.

        \column{0.55\textwidth}
        \centering
        \includegraphics[width=\linewidth]{exhibits/fig_temporal_drift_lift_testyear.pdf}
    \end{columns}
\end{frame}
\note{\textbf{[~2 min]} This is the counter-intuitive finding: retraining actively hurts. By exposing the algorithm to the post-ordinance era --- where petitions nearly disappear --- the model stops penalizing aggressive spatial configurations because the current environment has stopped producing them. The frozen legacy model outperforms because it learned during a period of high institutional activity. This has direct implications for deployment: a planner using a continually retrained model would lose predictive power precisely as the policy environment stabilizes.}

% ===== SLIDE 15: Agency in Temporal Shifts --- 2022 Dip (~2 min) =====
\begin{frame}{Agency in Temporal Shifts: What the 2022 Dip Tells Us}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{The 2022 Performance Pattern}
        \begin{itemize}
            \item \textbf{Finding}: PR-AUC drops sharply across all architectures in the 2022 evaluation window.
            \item \textbf{Interpretation}: Consistent with pandemic-era filing disruptions (2020--2021) reaching the evaluation window, combined with administrative uncertainty during the Land Development Code revision period.
            \item \textbf{Not a causal claim}: This is descriptive out-of-sample variation. No single institutional break is identified.
        \end{itemize}
        \vspace{0.8em}
        \textbf{What this implies about mobilization:}

        Predictive reliability is sensitive to ambient social shocks --- \textit{consistent with} petition mobilization being a strategic social choice, not a mechanical response to parcel spatiality.

        \column{0.5\textwidth}
        \centering
        \includegraphics[width=\linewidth]{exhibits/fig_temporal_drift_prauc_testyear.pdf}
        \vspace{0.5em}

\end{frame}
\note{\textbf{[~2 min]} Be explicit that ``consistent with'' is not ``proves.'' The thesis does not identify the causal mechanism behind the 2022 dip. What it shows is that model performance is not static across time, which is \textit{incompatible} with a fully mechanical view of opposition. The threshold sensitivity appendix (which replicates the drift analysis across 0\%, 5\%, 10\%, 15\%, 20\%, 25\% petition thresholds) shows that predictive signal does not stabilize around the statutory cutoff --- further evidence of the contingent social nature of mobilization.}

        \begin{itemize}
            \item \textit{``Haters are overrepresented and silent supporters are underrepresented.''} (Beaudet)
            \item Rigid cutoffs fail to \textit{``identify where people that are in actual need actually are.''} (Torrado)
            \item Professionals view probability thresholds as \textit{``ethically hazardous.''} (Tomko)
        \end{itemize}

        \column{0.4\textwidth}
        \centering

The motive blindness problem is not solvable with more data or better algorithms: it is a fundamental limit of predicting opposition from administrative records. The asymmetric transparency finding (Tomko) is important: formal opposition data over-represents loud voices because participation carries social costs, so the observed outcome variable is itself a biased sample of true community sentiment.}

% ===== SLIDE 17: Limitations (~2.5 min) =====
\begin{frame}{Limitations}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{Methodological}
        \begin{itemize}
            \item \textbf{Selection bias / chilling effects}: Only cases that entered the formal process are observed. Developers who anticipated opposition and withdrew or heavily modified proposals are invisible to the model. Attempted IPW failed overlap diagnostics and was discarded.
            \item \textbf{Timestamp imputation}: Milestone dates back-calculated from statutory minimums introduce measurement error in the feature matrix.
            \item \textbf{Unobserved clerk certification}: The outcome is a \textit{reconstructed} area share, not the clerk's legal determination.
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Ethical and Governance}
        \begin{itemize}
            \item \textbf{Proxy discrimination}: Even without protected-class inputs, reliance on neighborhood income and demographics encodes class and race dynamics --- a permanent methodological limitation, not a solved design feature.
            \item \textbf{Motive blindness}: Cannot separate exclusionary NIMBYism from legitimate anti-displacement resistance.
            \item \textbf{External validity}: Parameters are Austin-specific and bounded to the pre-HB~24 policy window. Methodology is portable; specific weights are not.
        \end{itemize}
    \end{columns}
\end{frame}
\note{\textbf{[~2.5 min]} Address limitations proactively before the committee raises them. The chilling-effect / selection bias limitation is the most substantive: the model is predicting on an already-censored subset of the decision space. The proxy discrimination concern is permanent and cannot be fully resolved by excluding protected-class features --- covariate correlates remain. Be direct: the study does not claim to have solved these problems; it maps them as boundaries on the findings.}

    \textbf{What surprised us}
    \begin{itemize}
        \item \textbf{Signal Dilution Paradox}: Retraining on modern data degrades lift; legacy models preserve spatial sorting intelligence learned during an active institutional era.
        \item \textbf{Neighborhood Agency}: Performance ceilings and erratic threshold sensitivity are \textit{consistent with} mobilization being a \textbf{strategic social choice}, not a mechanical administrative response.
    \end{itemize}

    \textbf{What it implies}
    \begin{itemize}
        \item Predictions belong in \textbf{relative triage rankings}, not case-level probability thresholds.
        \item Methodology (temporal holdouts, semantic clustering, qualitative integration) is \textbf{portable}; Austin-specific parameters are not.
    \end{itemize}
